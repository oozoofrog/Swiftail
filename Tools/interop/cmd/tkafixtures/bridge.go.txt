// Calls the actual tkaFilterNetmapLocked without duplicating the filtering logic (including rotationTracker).
package ipnlocal

import (
	"encoding/hex"
	"encoding/json"
	"fmt"
	"sort"

	"tailscale.com/health"
	"tailscale.com/tailcfg"
	"tailscale.com/tka"
	"tailscale.com/types/key"
	"tailscale.com/types/logger"
	"tailscale.com/types/netmap"
	"tailscale.com/util/eventbus"
)

type swiftailFixturePeer struct {
	NodeKey             string `json:"nodeKey"`
	Signature           string `json:"signature"`
	UnsignedPeerAPIOnly bool   `json:"unsignedPeerAPIOnly"`
}

type swiftailFixturePeerSet struct {
	AUMs  []string              `json:"aums"`
	Peers []swiftailFixturePeer `json:"peers"`
	Runs  int                   `json:"runs"`
}

func swiftailFixtureAUM(h string) (tka.AUM, error) {
	var a tka.AUM
	b, err := hex.DecodeString(h)
	if err != nil {
		return a, err
	}
	return a, a.Unserialize(b)
}

// SwiftailFixturePeerSets runs tkaFilterNetmapLocked Runs times per peer set and returns the distinct lists
// of retained peers' node keys (hexadecimal) in sorted order (rotationTracker iterates over a map, so outcomes can vary).
func SwiftailFixturePeerSets(input []byte) ([]byte, error) {
	var sets []swiftailFixturePeerSet
	if err := json.Unmarshal(input, &sets); err != nil {
		return nil, err
	}
	bus := eventbus.New()
	defer bus.Close()
	out := make([][][]string, len(sets))
	for i, set := range sets {
		seen := map[string][]string{}
		for range max(set.Runs, 1) {
			kept, err := swiftailFixtureFilter(bus, set)
			if err != nil {
				return nil, err
			}
			line, err := json.Marshal(kept)
			if err != nil {
				return nil, err
			}
			seen[string(line)] = kept
		}
		lines := make([]string, 0, len(seen))
		for line := range seen {
			lines = append(lines, line)
		}
		sort.Strings(lines)
		for _, line := range lines {
			out[i] = append(out[i], seen[line])
		}
	}
	return json.Marshal(out)
}

func swiftailFixtureFilter(bus *eventbus.Bus, set swiftailFixturePeerSet) ([]string, error) {
	storage := tka.ChonkMem()
	genesis, err := swiftailFixtureAUM(set.AUMs[0])
	if err != nil {
		return nil, err
	}
	authority, err := tka.Bootstrap(storage, genesis)
	if err != nil {
		return nil, err
	}
	if len(set.AUMs) > 1 {
		updates := make([]tka.AUM, len(set.AUMs)-1)
		for j, h := range set.AUMs[1:] {
			if updates[j], err = swiftailFixtureAUM(h); err != nil {
				return nil, err
			}
		}
		if err := authority.Inform(storage, updates); err != nil {
			return nil, err
		}
	}
	nm := &netmap.NetworkMap{}
	for j, p := range set.Peers {
		var nk key.NodePublic
		if err := nk.UnmarshalText([]byte("nodekey:" + p.NodeKey)); err != nil {
			return nil, err
		}
		sig, err := hex.DecodeString(p.Signature)
		if err != nil {
			return nil, err
		}
		n := &tailcfg.Node{ID: tailcfg.NodeID(j + 1), StableID: tailcfg.StableNodeID(fmt.Sprintf("n%d", j+1)), Key: nk, KeySignature: sig, UnsignedPeerAPIOnly: p.UnsignedPeerAPIOnly}
		nm.Peers = append(nm.Peers, n.View())
	}
	b := &LocalBackend{logf: logger.Discard, health: health.NewTracker(bus), tka: &tkaState{authority: authority, storage: storage}}
	b.tkaFilterNetmapLocked(nm)
	kept := []string{}
	for _, p := range nm.Peers {
		raw := p.Key().Raw32()
		kept = append(kept, hex.EncodeToString(raw[:]))
	}
	return kept, nil
}
