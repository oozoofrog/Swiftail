// Calls tailscale.com v1.104.6 ipn/ipnlocal/local.go and node_backend.go (BSD-3-Clause).
// Captures the configuration that the actual LocalBackend.authReconfigLocked passes to Engine.Reconfig.
package ipnlocal

import (
	"errors"
	"tailscale.com/ipn"
	"tailscale.com/ipn/store/mem"
	"tailscale.com/net/dns"
	"tailscale.com/tsd"
	"tailscale.com/types/logger"
	"tailscale.com/types/logid"
	"tailscale.com/types/netmap"
	"tailscale.com/types/persist"
	"tailscale.com/wgengine"
	"tailscale.com/wgengine/router"
	"tailscale.com/wgengine/wgcfg"
)

type swiftailCaptureEngine struct {
	wgengine.Engine
	dns    *dns.Config
	router *router.Config
}

func (e *swiftailCaptureEngine) Reconfig(_ *wgcfg.Config, r *router.Config, d *dns.Config) error {
	e.dns, e.router = d, r
	return nil
}

// SwiftailFixtureBackend is a test node that accepts only a fixed netmap, with no control connection.
type SwiftailFixtureBackend struct {
	b   *LocalBackend
	e   *swiftailCaptureEngine
	sys *tsd.System
}

// NewSwiftailFixtureBackend uses the same in-memory store and fake TUN as the upstream LocalBackend tests.
func NewSwiftailFixtureBackend() (*SwiftailFixtureBackend, error) {
	sys := tsd.NewSystem()
	sys.Set(new(mem.Store))
	engine, err := wgengine.NewFakeUserspaceEngine(logger.Discard, sys.Set, sys.HealthTracker.Get(), sys.UserMetricsRegistry(), sys.Bus.Get())
	if err != nil {
		sys.Bus.Get().Close()
		return nil, err
	}
	e := &swiftailCaptureEngine{Engine: engine}
	sys.Set(e)
	dialer := sys.Dialer.Get()
	b, err := NewLocalBackend(logger.Discard, logid.PublicID{}, sys, 0)
	if err != nil {
		engine.Close()
		dialer.Close()
		sys.Bus.Get().Close()
		return nil, err
	}
	b.goos = "ios"
	return &SwiftailFixtureBackend{b, e, sys}, nil
}

// Capture runs the actual reconfiguration without duplicating DNS or router configuration logic.
func (f *SwiftailFixtureBackend) Capture(nm *netmap.NetworkMap, expired bool) (*dns.Config, *router.Config, error) {
	b := f.b
	b.mu.Lock()
	defer b.mu.Unlock()
	p := ipn.NewPrefs()
	p.WantRunning = true
	p.CorpDNS = true
	p.RouteAll = false
	p.Persist = new(persist.Persist)
	if err := b.pm.setPrefsNoPermCheck(p.View()); err != nil {
		return nil, nil, err
	}
	b.currentNode().SetNetMap(nm)
	b.keyExpired = expired
	f.sys.ControlKnobs().UpdateFromNodeAttributes(nm.SelfNode.CapMap().AsMap())
	f.e.dns = nil
	f.e.router = nil
	b.authReconfigLocked()
	if f.e.dns == nil || f.e.router == nil {
		return nil, nil, errors.New("the configuration was not passed to the engine")
	}
	return f.e.dns, f.e.router, nil
}

// Close shuts down the node and all test resources.
func (f *SwiftailFixtureBackend) Close() {
	f.b.Shutdown()
	f.sys.Bus.Get().Close()
}
