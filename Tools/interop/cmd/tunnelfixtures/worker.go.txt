const suffix = "tail0001.ts.net"
const fixedNow = "2026-01-02T03:04:05Z"

type node struct {
	ID         int            `json:"id"`
	Name       string         `json:"name"`
	Addresses  []netip.Prefix `json:"addresses"`
	AllowedIPs []netip.Prefix `json:"allowedIPs"`
	Expired    bool           `json:"expired"`
	KeyExpiry  string         `json:"keyExpiry,omitempty"`
	HomeDERP   int            `json:"homeDERP"`
}
type input struct {
	Self        node               `json:"self"`
	Peers       []node             `json:"peers"`
	DNS         tailcfg.DNSConfig  `json:"dnsConfig"`
	CapMap      tailcfg.NodeCapMap `json:"capMap"`
	RouteAll    bool               `json:"routeAll"`
	ExitNode    string             `json:"exitNode"`
	SelfExpired bool               `json:"selfExpired"`
	MTU         int                `json:"mtu"`
}
type dnsOutput struct {
	Servers       []netip.Addr `json:"servers"`
	MatchDomains  []string     `json:"matchDomains"`
	SearchDomains []string     `json:"searchDomains"`
	LocalDomains  []string     `json:"localDomains"`
	Error         string       `json:"error,omitempty"`
}
type output struct {
	Addresses  []netip.Prefix `json:"addresses"`
	PeerRoutes []netip.Prefix `json:"peerRoutes"`
	Routes     []netip.Prefix `json:"routes"`
	DNS        dnsOutput      `json:"dns"`
	DNSConfig  *dns.Config    `json:"dnsConfig"`
}
type settings struct {
	Addresses       []netip.Prefix `json:"addresses"`
	Routes          []netip.Prefix `json:"routes"`
	DNSServers      []netip.Addr   `json:"dnsServers"`
	DNSMatchDomains []string       `json:"dnsMatchDomains"`
	MTU             int            `json:"mtu"`
}
type step struct {
	Name            string    `json:"name"`
	Input           input     `json:"input"`
	Go              output    `json:"go"`
	Settings        *settings `json:"settings"`
	SwiftailDiffers string    `json:"swiftailDiffers,omitempty"`
}
type scenario struct {
	Name  string `json:"name"`
	Steps []step `json:"steps"`
}

func prefixes(ss ...string) []netip.Prefix {
	out := []netip.Prefix{}
	for _, s := range ss {
		out = append(out, netip.MustParsePrefix(s))
	}
	return out
}
func base() input {
	return input{Self: node{ID: 1, Name: "rd-mac-1a2b3c4d." + suffix + ".", Addresses: prefixes("100.64.0.1/32", "fd7a:115c:a1e0::1/128"), AllowedIPs: prefixes("100.64.0.1/32", "fd7a:115c:a1e0::1/128"), HomeDERP: 1}, Peers: []node{}, DNS: tailcfg.DNSConfig{Proxied: true, Domains: []string{suffix}}, CapMap: tailcfg.NodeCapMap{}, MTU: 1280}
}
func peer(id int, addresses ...string) node {
	p := prefixes(addresses...)
	return node{ID: id, Name: fmt.Sprintf("rd-peer-%08x.%s.", id, suffix), Addresses: p, AllowedIPs: slices.Clone(p), HomeDERP: 1}
}
func one(name string, in input, reason string) scenario {
	return scenario{Name: name, Steps: []step{{Name: "Settings", Input: in, SwiftailDiffers: reason}}}
}
func scenarios() []scenario {
	out := []scenario{}
	for _, family := range []string{"IPv4", "IPv6", "Dual-stack"} {
		in := base()
		if family == "IPv4" {
			in.Self.Addresses = prefixes("100.64.0.1/32")
		}
		if family == "IPv6" {
			in.Self.Addresses = prefixes("fd7a:115c:a1e0::1/128")
		}
		in.Self.AllowedIPs = slices.Clone(in.Self.Addresses)
		for _, n := range []int{0, 1, 3} {
			in.Peers = []node{}
			for i := 0; i < n; i++ {
				in.Peers = append(in.Peers, peer(i+2, fmt.Sprintf("100.64.0.%d/32", i+2), fmt.Sprintf("fd7a:115c:a1e0::%x/128", i+2)))
			}
			reason := "D3: Omit search domains"
			if family == "IPv4" {
				reason += ". S4 and S5: Omit IPv6 routes and DNS servers when self has no IPv6 address"
			}
			if family == "IPv6" && n > 0 {
				reason += ". S4: Omit IPv4 routes when self has no IPv4 address"
			}
			out = append(out, one(family+fmt.Sprintf(" peers %d", n), in, reason))
		}
	}
	in := base()
	in.DNS.Domains = nil
	in.Peers = []node{peer(2, "100.64.0.2/32")}
	out = append(out, one("IPv4 peers without search domains", in, ""))
	in = base()
	in.DNS.Domains = nil
	in.Peers = []node{peer(2, "fd7a:115c:a1e0::2/128")}
	out = append(out, one("Aggregate one ULA peer to /48", in, ""))
	s := scenario{Name: "OneCGNAT threshold and control directive", Steps: []step{}}
	for _, mode := range []string{"none", "true", "false"} {
		for _, n := range []int{1, 2} {
			in = base()
			in.DNS.Domains = nil
			if mode != "none" {
				in.CapMap["one-cgnat?v="+nodecap.Cap(mode)] = nil
			}
			for i := 0; i < n; i++ {
				in.Peers = append(in.Peers, peer(i+2, fmt.Sprintf("100.64.0.%d/32", i+2)))
			}
			s.Steps = append(s.Steps, step{Name: fmt.Sprintf("Directive %s, %d peers", mode, n), Input: in})
		}
	}
	out = append(out, s)
	in = base()
	in.DNS.Domains = nil
	p := peer(2, "100.64.0.2/32", "fd7a:115c:a1e0::2/128")
	p.Expired = true
	p.KeyExpiry = "2026-01-01T00:00:00Z"
	in.Peers = []node{p}
	out = append(out, one("Omit routes to expired peers", in, ""))
	in = base()
	in.DNS.Domains = nil
	p = peer(2, "100.64.0.2/32")
	p.KeyExpiry = "2026-01-01T00:00:00Z"
	in.Peers = []node{p}
	out = append(out, one("Past KeyExpiry with Expired false", in, ""))
	in = base()
	in.DNS.Domains = nil
	p = peer(2, "100.64.0.2/32")
	p.HomeDERP = 0
	in.Peers = []node{p}
	out = append(out, one("Peer without disco or DERP", in, ""))
	in = base()
	in.DNS.Domains = nil
	p = peer(2, "100.64.0.2/32", "fd7a:115c:a1e0::2/128")
	p.AllowedIPs = append(p.AllowedIPs, prefixes("192.0.2.0/24", "0.0.0.0/0", "::/0")...)
	in.Peers = []node{p}
	out = append(out, one("Ignore advertised subnet and exit routes", in, ""))
	in = base()
	in.DNS.Proxied = false
	out = append(out, one("MagicDNS disabled", in, "D3: Omit search domains"))
	in = base()
	in.Self.Name = ""
	in.DNS.Domains = nil
	out = append(out, one("No MagicDNS suffix", in, ""))
	in = base()
	in.DNS.Proxied = false
	in.DNS.Domains = nil
	in.DNS.Routes = map[string][]*dnstype.Resolver{"local.tail0001.ts.net": {}}
	out = append(out, one("Empty Routes mark a local zone", in, "D1: Do not configure the primary resolver even when the node index contains out-of-zone names"))
	in = base()
	in.DNS.Domains = nil
	in.DNS.Routes = map[string][]*dnstype.Resolver{"split.example.com": {{Addr: "192.0.2.53"}}}
	out = append(out, one("Custom split DNS", in, "D2: Ignore Routes with resolvers and upstream forwarding, and match only local zones"))
	in = base()
	in.DNS.Domains = nil
	in.DNS.Resolvers = []*dnstype.Resolver{{Addr: "192.0.2.53"}}
	out = append(out, one("Global resolvers", in, "D1 and D2: Ignore global resolvers and match only local zones"))
	in = base()
	in.DNS.Domains = nil
	in.DNS.Proxied = false
	in.DNS.Resolvers = []*dnstype.Resolver{{Addr: "192.0.2.53"}}
	out = append(out, one("Global resolvers only", in, "D1 and D2: Omit DNS when there are no local zones"))
	in = base()
	in.DNS.Domains = nil
	in.DNS.ExtraRecords = []tailcfg.DNSRecord{{Name: "rd-extra.example.com", Type: "A", Value: "100.64.0.9"}}
	out = append(out, one("ExtraRecords without overrides", in, ""))
	in = base()
	in.DNS.Domains = nil
	in.DNS.ExtraRecords = []tailcfg.DNSRecord{{Name: "rd-extra.example.com", Value: "100.64.0.9"}}
	in.DNS.Routes = map[string][]*dnstype.Resolver{"example.com": {{Addr: "192.0.2.53"}}}
	out = append(out, one("Custom route overrides ExtraRecords", in, "D2: Ignore custom split DNS and upstream forwarding"))
	in = base()
	in.DNS.Domains = nil
	in.CapMap[nodecap.ForceRegisterMagicDNSIPv4Only] = nil
	out = append(out, one("Directive to register IPv4 DNS only", in, ""))
	in = base()
	in.SelfExpired = true
	in.Self.Expired = true
	in.Self.KeyExpiry = "2026-01-01T00:00:00Z"
	out = append(out, one("No DNS when the self key expires", in, ""))
	in = base()
	in.Self.Addresses = []netip.Prefix{}
	in.Self.AllowedIPs = []netip.Prefix{}
	out = append(out, one("No self addresses", in, "S1: Omit settings when self has no addresses"))
	return out
}
func convert(n node) *tailcfg.Node {
	p := &tailcfg.Node{ID: tailcfg.NodeID(n.ID), StableID: tailcfg.StableNodeID(fmt.Sprintf("fixture-%d", n.ID)), Name: n.Name, Addresses: n.Addresses, AllowedIPs: n.AllowedIPs, Expired: n.Expired, HomeDERP: tailcfg.DERPRegionID(n.HomeDERP), MachineAuthorized: true}
	// Use only public test identifiers. Do not create private keys or include them in the fixture.
	if err := p.Key.UnmarshalText([]byte(fmt.Sprintf("nodekey:%064x", n.ID))); err != nil {
		panic(err)
	}
	if n.KeyExpiry != "" {
		var err error
		p.KeyExpiry, err = time.Parse(time.RFC3339, n.KeyExpiry)
		if err != nil {
			panic(err)
		}
	}
	return p
}

type osCapture struct{ cfg dns.OSConfig }

func (o *osCapture) SetDNS(c dns.OSConfig) error { o.cfg = c; return nil }
func (*osCapture) SupportsSplitDNS() bool        { return true }

// Do not read actual OS DNS. Provide fake defaults so primary mode also compiles deterministically.
func (*osCapture) GetBaseConfig() (dns.OSConfig, error) {
	return dns.OSConfig{Nameservers: []netip.Addr{netip.MustParseAddr("192.0.2.54")}}, nil
}
func (*osCapture) Close() error { return nil }
func domains(v []dnsname.FQDN) []string {
	out := []string{}
	for _, s := range v {
		out = append(out, strings.TrimSuffix(string(s), "."))
	}
	return out
}
func compile(cfg dns.Config, caps tailcfg.NodeCapMap) (dnsOutput, error) {
	bus := eventbus.New()
	defer bus.Close()
	tracker := health.NewTracker(bus)
	dialer := tsdial.NewDialer(netmon.NewStatic())
	dialer.SetBus(bus)
	defer dialer.Close()
	knobs := new(controlknobs.Knobs)
	knobs.UpdateFromNodeAttributes(caps)
	oscfg := new(osCapture)
	m := dns.NewManager(logger.Discard, oscfg, tracker, dialer, nil, knobs, "ios", bus)
	defer m.Down()
	local := []string{}
	m.Resolver().TestOnlySetHook(func(c resolver.Config) { local = domains(c.LocalDomains) })
	err := m.Set(cfg)
	out := dnsOutput{Servers: append([]netip.Addr{}, oscfg.cfg.Nameservers...), MatchDomains: domains(oscfg.cfg.MatchDomains), SearchDomains: domains(oscfg.cfg.SearchDomains), LocalDomains: local}
	if err != nil {
		out.Error = err.Error()
	}
	return out, err
}
func family(p netip.Prefix, v4, v6 bool) bool { return p.Addr().Is4() && v4 || p.Addr().Is6() && v6 }
func evaluate(in input, backend *ipnlocal.SwiftailFixtureBackend) (output, *settings, error) {
	self := convert(in.Self)
	self.CapMap = in.CapMap
	nm := &netmap.NetworkMap{SelfNode: self.View(), DNS: in.DNS, AllCaps: set.Set[nodecap.Cap]{}}
	for c := range in.CapMap {
		nm.AllCaps.Add(c)
	}
	rm := routemanager.New(logger.Discard)
	mutation := rm.Begin()
	knobs := new(controlknobs.Knobs)
	knobs.UpdateFromNodeAttributes(in.CapMap)
	one, _ := knobs.OneCGNAT.Load().Get()
	mutation.SetTailnetConfig(routemanager.TailnetConfig{OneCGNAT: one})
	mutation.SetPrefs(routemanager.Prefs{RouteAll: false})
	for _, n := range in.Peers {
		p := convert(n)
		nm.Peers = append(nm.Peers, p.View())
		mutation.UpsertPeer(p.View())
	}
	mutation.Commit()
	peerRoutes := []netip.Prefix{}
	for p := range rm.OSRoutes().All() {
		peerRoutes = append(peerRoutes, p)
	}
	slices.SortFunc(peerRoutes, func(a, b netip.Prefix) int {
		if c := a.Addr().Compare(b.Addr()); c != 0 {
			return c
		}
		return a.Bits() - b.Bits()
	})
	dcfg, rcfg, err := backend.Capture(nm, in.SelfExpired)
	if err != nil {
		return output{}, nil, err
	}
	dnsOut, err := compile(*dcfg, in.CapMap)
	if err != nil {
		return output{}, nil, err
	}
	out := output{Addresses: append([]netip.Prefix{}, rcfg.LocalAddrs...), PeerRoutes: peerRoutes, Routes: append([]netip.Prefix{}, rcfg.Routes...), DNS: dnsOut, DNSConfig: dcfg}
	// Cross-check the public API path against the RouteManager result used by LocalBackend.
	rawRoutes := slices.Clone(peerRoutes)
	v4, v6 := false, false
	for _, p := range in.Self.Addresses {
		v4 = v4 || p.Addr().Is4()
		v6 = v6 || p.Addr().Is6()
	}
	if v4 {
		rawRoutes = append(rawRoutes, netip.PrefixFrom(tsaddr.TailscaleServiceIP(), 32))
	}
	if v6 {
		rawRoutes = append(rawRoutes, netip.PrefixFrom(tsaddr.TailscaleServiceIPv6(), 128))
	}
	if !slices.Equal(rawRoutes, out.Routes) {
		return output{}, nil, fmt.Errorf("router routes differ between the public RouteManager and LocalBackend")
	}
	result := &settings{Addresses: []netip.Prefix{}, Routes: []netip.Prefix{}, DNSServers: []netip.Addr{}, DNSMatchDomains: []string{}, MTU: in.MTU}
	for _, p := range out.Addresses {
		if p.IsSingleIP() {
			result.Addresses = append(result.Addresses, p)
		}
	}
	if len(result.Addresses) == 0 {
		return out, nil, nil
	}
	for _, p := range out.Routes {
		if family(p, v4, v6) {
			result.Routes = append(result.Routes, p)
		}
	}
	// Apply only the narrowing in behavior S5. Take the local zones from the Go Manager's actual resolver.Config.
	if len(dnsOut.LocalDomains) > 0 {
		if v4 {
			result.DNSServers = append(result.DNSServers, tsaddr.TailscaleServiceIP())
		}
		if v6 && !knobs.ShouldForceRegisterMagicDNSIPv4Only() {
			result.DNSServers = append(result.DNSServers, tsaddr.TailscaleServiceIPv6())
		}
		result.DNSMatchDomains = slices.Clone(dnsOut.LocalDomains)
	}
	return out, result, nil
}
func run() error {
	envknob.Setenv("TS_DEBUG_MAGIC_DNS_DUAL_STACK", "false")
	envknob.Setenv("TS_NO_LOGS_NO_SUPPORT", "true")
	envknob.Setenv("TS_DISABLE_PORTMAPPER", "true")
	backend, err := ipnlocal.NewSwiftailFixtureBackend()
	if err != nil {
		return err
	}
	defer backend.Close()
	ss := scenarios()
	steps := 0
	for i := range ss {
		for j := range ss[i].Steps {
			st := &ss[i].Steps[j]
			steps++
			st.Go, st.Settings, err = evaluate(st.Input, backend)
			if err != nil {
				return fmt.Errorf("%s/%s: %w", ss[i].Name, st.Name, err)
			}
		}
	}
	result := struct {
		Reference     string     `json:"reference"`
		ScenarioCount int        `json:"scenarioCount"`
		StepCount     int        `json:"stepCount"`
		Now           string     `json:"now"`
		BaseResolvers []string   `json:"baseResolvers"`
		Scenarios     []scenario `json:"scenarios"`
	}{"tailscale.com v1.104.6", len(ss), steps, fixedNow, []string{"192.0.2.54"}, ss}
	enc := json.NewEncoder(os.Stdout)
	enc.SetIndent("", "  ")
	return enc.Encode(result)
}
func main() {
	if err := run(); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
